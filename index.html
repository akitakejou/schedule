function generate() { // 希望読み込み employees.forEach(emp => { emp.workLog = []; emp.lastNight = -100; for (let d = 1; d <= 7; d++) { emp.pref[d] = document.getElementById(emp.name + "_" + d).value || "free"; } }); let schedule = {}; for (let d = 1; d <= 7; d++) { schedule[d] = {}; for (let h = 0; h < 24; h++) { let candidates = []; for (let emp of employees) { // ---- ハード制約 ---- const pref = emp.pref[d]; if (pref === "off") continue; if (pref.includes("-")) { const [s,e] = pref.split("-").map(Number); if (h < s || h >= e) continue; } if (emp.type === "student" && h >= 20) continue; // 夜勤後2日休み if (d - emp.lastNight <= 2) continue; // ---- スコア ---- let score = 0; if (pref === "free") score += 5; if (pref.includes("-")) { const [s,e] = pref.split("-").map(Number); if (h >= s && h < e) score += 10; } // 偏り防止 score -= emp.workLog.length * 0.5; candidates.push({emp, score}); } // スコア順 candidates.sort((a,b)=>b.score-a.score); let needed = (h >= 22 || h < 8) ? 2 : 2; let assigned = []; // ① まず正社員 or 派遣を1人確保 let core = candidates.find(c => c.emp.type === "full" || c.emp.type === "temp"); if (core) { assigned.push(core.emp); } // ② 残り埋める for (let c of candidates) { if (assigned.length >= needed) break; if (!assigned.includes(c.emp)) { assigned.push(c.emp); } } // ③ 足りなければ強制投入（応援＝派遣） while (assigned.length < needed) { assigned.push({name:"応援", type:"temp"}); } // 記録 schedule[d][h] = assigned; assigned.forEach(emp => { if (emp.workLog) { emp.workLog.push({d,h}); if (h >= 22 || h < 8) { emp.lastNight = d; } } }); } } render(schedule); } function render(schedule) { let html = ""; for (let d=1; d<=7; d++) html += ""; html += ""; employees.forEach(emp => { html += ""; for (let d=1; d<=7; d++) { let hours = []; for (let h=0; h<24; h++) { if (schedule[d][h].find(e => e.name === emp.name)) { hours.push(h); } } let cls = ""; if (hours.some(h=>h>=22||h<8)) cls="night"; else if (hours.length>0) cls="free"; html += ``; } html += ""; }); html += "
名前"+d+"
"+emp.name+"${hours.length>0 ? "出" : "休"}
"; document.getElementById("result").innerHTML = html; }
